# i have done in the repository https://github.com/Dicksonsilayo/AmountTrackerChallengeInCLI.git
# usage 
# to add expense you should type php expenses.php add "the description of your expense" "the amount"
# to delete an expense php expenses.php delete "the id of the expense to delete"
# to view all the expenses php expenses.php viewAll
# to view monthly expenses php expenses.php month "the month eg 06 for june"
# update an expense php expenses.php update "id of the expense" "new description" "new amount"
# to filter expenses by date php expenses.php filterFrom "startdate eg 24-06-2026" "destination date eg 27-07-2026"
